% ===================== 第一部分：Why & What =====================
\section{开篇与背景：The ``Why'' \& ``What''}

\begin{frame}{IT 生产模式的“内卷”困境}
  \begin{itemize}
    \item 需求堆积、交付周期冗长，排期越来越难。
    \item 大量重复性工作（CRUD、告警处理、脚本维护）消耗研发人力。
    \item 跨部门沟通成本高，信息孤岛明显，文档分散。
    \item 技术栈快速迭代，学习与迁移成本不断上升。
    \item 核心问题：如何从“堆人力”转向“堆算力 + 智能”的新生产方式？
  \end{itemize}
\end{frame}

\begin{frame}{概念重构：LLM、Copilot 与 AI Agent}
  \begin{itemize}
    \item \textbf{LLM：} 会“思考”但不直接执行——主要提供语言理解与生成。
    \item \textbf{Copilot：} 人主导、AI 辅助——补全代码、给建议，但不闭环。
    \item \textbf{AI Agent：} 目标驱动的自主智能体，具备：
      \begin{itemize}
        \item 感知（Perception）——读取代码库、日志、文档。
        \item 规划（Planning）——拆解任务、生成多步计划。
        \item 记忆（Memory）——跨对话、跨文件保持上下文。
        \item 工具使用（Tool Use）——调用 CLI、API、脚本，自动执行闭环。
      \end{itemize}
    \item 标准 Agent Pipeline：LLM + Planning + Memory + Tools。
  \end{itemize}
\end{frame}

\begin{frame}{从问答机器人到垂直领域机器人}
  \begin{itemize}
    \item 最开始的使用方式：普通问答机器人，类似“智能搜索 + 聊天”。
    \item 进阶：在 Poe 等平台设置语料库，构建特定垂直领域机器人。
    \item 感受：比搜索引擎更懂上下文，但仍然停留在“我问它答”的阶段。
    \item 引子：AI 是否只能做问答？能否真正作为 \textbf{Agent} 主动执行任务？
  \end{itemize}
\end{frame}

\section{实战案例：长文出版 \& 工作流自动化}

\begin{frame}{案例：AI 写长文 Report 的痛点}
  \begin{itemize}
    \item 场景：基于既有白皮书迭代新版，同时要出英文、繁中、简中三种版本。
    \item 痛点 1：PDF 理解能力有限、Token 消耗高。
    \item 痛点 2：十万字级长文本翻译，AI 容易中断、风格不统一。
    \item 痛点 3：LaTeX 排版 & Debug 成本高，人工改格式很耗时。
    \item 痛点 4：一处内容更新，需要手动在三种语言版本中同步，非常繁琐。
  \end{itemize}
\end{frame}

\begin{frame}{我的解决方案：从 PDF 到自动发布}
  \begin{itemize}
    \item 第一步：将 PDF 转为纯文本（如 \texttt{.md}），再喂给大模型：
      \begin{itemize}
        \item 文本比 PDF 更容易、更便宜让模型理解。
        \item Token 消耗大约可降到 PDF 的 $1/2 \sim 1/20$。
      \end{itemize}
    \item 第二步：在原始文本上“覆盖式”编辑，而不是单独生成新文档。
    \item 第三步：让 AI Agent 直接操作 LaTeX 源码，协助排版与 Debug。
    \item 第四步：用大模型 API + 脚本 + GitHub Actions：
      \begin{itemize}
        \item 自动翻译多语言版本。
        \item 自动 LaTeX 编译、网站构建与部署。
        \item 自己几乎不需要手写代码即可完成。
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{思考一：给 AI “降维”——文本 vs PDF}
  \begin{itemize}
    \item \textbf{文本文件（.md/.tex）}：
      \begin{itemize}
        \item 没有复杂格式和 OCR 噪音，结构更清晰。
        \item 模型更容易抽取章节结构、公式、列表。
      \end{itemize}
    \item \textbf{PDF}：
      \begin{itemize}
        \item 文本型 PDF 还好，扫描版需依赖 OCR，容易出错。
        \item 上传到 Poe.com 时，效果取决于模型和排版质量。
      \end{itemize}
    \item \textbf{记忆模式}：
      \begin{itemize}
        \item 上下文记忆：仅当次对话有效。
        \item 训练记忆：模型固化的知识，无法被对话直接更新。
        \item 目前没有“长期个人记忆”，所以要靠脚本和仓库来“留住”成果。
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{思考二：脚本化 = 更适合 AI Agent 的场景}
  \begin{itemize}
    \item Word 文档：难以完全脚本化；格式操作对机器不友好。
    \item LaTeX：从内容、排版到编译，全流程都可用命令行控制，天然适合 Agent。
    \item 类推：做 PPT 是否也能脚本化？
      \begin{itemize}
        \item 本次分享的 PPT 就是用 LaTeX + AI Agent 完成的。
        \item 我主要专注内容；版式、格式、细节交给 AI。
      \end{itemize}
    \item 结论：\textbf{能脚本化的任务，更容易被 AI Agent 高效接管。}
  \end{itemize}
\end{frame}

\begin{frame}{思考三：架构思维 \& Domain Knowledge 的价值}
  \begin{itemize}
    \item 工程实现需要大量“用户手册 + 实战经验”，例如：
      \begin{itemize}
        \item CI/CD Workflow、YAML 语法、部署策略等。
      \end{itemize}
    \item 外行人描述需求：只会说“帮我做个网站 / 小程序”。
    \item 内行拆解任务：前端、后端、测试、运维、日志、监控……每一步可交给 Agent。
    \item 设计师知道“毛玻璃特效”，PPT 设计师知道“分镜结构”：
      \begin{itemize}
        \item 这些专业词汇让你能更精确地驱动 AI。
      \end{itemize}
    \item 所以：\textbf{AI 时代，Domain Knowledge 不会被取代，反而更重要。}
  \end{itemize}
\end{frame}

\section{进阶技巧：Prompt 套娃、二创与个人知识库}

\begin{frame}{Prompt “套娃”：让 AI 帮你写 Prompt}
  \begin{itemize}
    \item 原始需求：\textbf{“请帮我写一个网页，加入工业风金属元素，高端风格。”}
    \item 套娃思路：
      \begin{itemize}
        \item 把这句简单需求喂给 AI。
        \item 告诉它：你是外行，希望它以“硅谷科技公司顶级 UI + 前端工程师”的身份，\textbf{帮你升级这个 Prompt}。
      \end{itemize}
    \item 结果：AI 生成了一个极其专业的 Prompt：
      \begin{itemize}
        \item 描述工业风视觉语言、材质（拉丝金属、磨砂金属、冷灰色系）。
        \item 说明栅格系统、信息架构、动效细节、HTML/CSS 结构建议。
      \end{itemize}
    \item 小结：一次简单的“套娃”，你就获得了接近 Apple/Tesla 级别的设计说明书。
  \end{itemize}
\end{frame}

\begin{frame}{个人知识库：从博主到“专属智能体”}
  \begin{itemize}
    \item 场景：你很喜欢一个博主 / 技术宅 / 恋爱高手，但他的视频太多、没时间刷完。
    \item 做法：
      \begin{itemize}
        \item 用 \url{http://glasp.co/} 抓取视频字幕，一键复制文本。
        \item 把这些字幕整理成语料库，交给大模型进行分析和问答。
      \end{itemize}
    \item 效果：
      \begin{itemize}
        \item 为某个博主“生成一个智能体”，可以随时询问他的观点与方法论。
        \item \textbf{glasp-subtitle.png} 可作为实际操作界面示例截图。
      \end{itemize}
  \end{itemize}

  \vspace{0.5em}
  \centering
  \includegraphics[width=0.6\textwidth]{./pictures/glasp-subtitle.png}\\[4pt]
  \small glasp-subtitle.png
\end{frame}

\begin{frame}{NotebookLM：从语料到播客与幻灯片}
  \begin{itemize}
    \item NotebookLM 是一个面向个人知识管理的神器：
      \begin{itemize}
        \item 可以根据上传的语料（文本、PDF、视频链接）生成播客、报告、幻灯片。
      \end{itemize}
    \item 使用技巧：
      \begin{itemize}
        \item 生成的幻灯片通常以 PDF 形式导出，审美不错（类似 Nano Banana 风格）。
        \item 想继续编辑，可用 WPS 将 PDF 转成 \texttt{.pptx}，再进行微调。
      \end{itemize}
    \item 我尤其喜欢播客功能：
      \begin{itemize}
        \item 跑步、做家务、躺着休息时都能“听”自己的知识库。
        \item 对于自媒体创作者，这是极大提升生产力的工具。
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{二创时代：AI 让“模仿 +创新”更高效}
  \begin{itemize}
    \item 现实中：二次创作者在小红书、B 站经常比原作者获得更多流量。
      \begin{itemize}
        \item 小米 SU7 被认为向保时捷学习外观，但大众非常买账。
      \end{itemize}
    \item 过去没有 AI 时，二创依然门槛很高：剪辑、撰稿、排版都很耗时。
    \item AI 时代：二创效率至少提高一个数量级：
      \begin{itemize}
        \item 快速整理原始素材、提炼观点、生成新表达。
        \item 辅助生成封面、标题、文案、脚本。
      \end{itemize}
    \item 摄影 / 影像创作延伸：
      \begin{itemize}
        \item “摄影大赛”建议更名为“AI 摄影大赛”——给两张照片作为首帧和尾帧，就能生成一段视频。
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Claude CLI 常用指令与工作流（补充）}
  \begin{itemize}
    \item 常见用法示例：
      \begin{itemize}
        \item \texttt{claude}：普通对话模式。
        \item \texttt{claude code}：侧重代码理解、重构、测试生成。
        \item \texttt{claude chat .}：在当前项目目录中开启上下文对话。

      \begin{itemize}

              在“完全信任”的仓库中自动执行命令：
          \begin{itemize}
            \item 例如：“喝一杯咖啡的时间”，让它从架构设计到部署测试跑完一整套流程。
          \end{itemize}
        \item 建议：仅在熟悉且可随时回滚的项目中使用“dangerously”选项。
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{上下文记忆与 CLAUDE.md（补充）}
  \begin{itemize}
    \item 上下文记忆如何把握？
      \begin{itemize}
        \item 把长任务拆成若干可在单次对话内完成的步骤。
        \item 经常让 Agent 用“要点总结”压缩当前进展，减少 Token 压力。
      \end{itemize}
    \item \textbf{CLAUDE.md} 是什么？
      \begin{itemize}
        \item 放在仓库根目录的说明文件，类似项目级 “使用说明 + 约束”。
        \item 内容包括：项目背景、技术栈、目录结构、代码风格、约定、禁用操作等。
        \item 好处：
          \begin{itemize}
            \item 每次对话都可自动被读取，减少重复解释，\textbf{间接节约 Token}。
            \item 保证不同对话中行为一致，比如统一日志风格、错误处理策略。
          \end{itemize}
      \end{itemize}
    \item 对应 Gemini 等：
      \begin{itemize}
        \item 也可以通过配置文件或系统 Prompt 方式，扮演“项目说明书”的角色。
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{/subagent、/commands、skills 与 MCP（补充）}
  \begin{itemize}
    \item \textbf{/subagent}：
      \begin{itemize}
        \item 在同一个会话中创建多个“子代理”，例如“测试工程师”、“安全审计员”、“性能优化师”。
        \item 适合复杂任务的多角色协作，例如大型重构或安全评估。
      \end{itemize}
    \item \textbf{/commands \& skills}：
      \begin{itemize}
        \item 定义常用命令/技能的快捷方式，如“一键生成测试用例”、“一键整理 Changelog”。
        \item 相当于给 Agent 配备一套“宏命令”工具栏。
      \end{itemize}
    \item \textbf{MCP（Model Context Protocol）}：
      \begin{itemize}
        \item 通过 MCP，Agent 可以安全地访问更多外部系统：
          \begin{itemize}
            \item 内部知识库、数据库、Issue Tracker、监控告警系统等。
          \end{itemize}
        \item 示例场景：
          \begin{itemize}
            \item 从监控系统取最近 1 小时 500 错误日志，自动分析根因。
            \item 读取 Jira 任务列表，为每个任务生成实现建议与测试清单。
          \end{itemize}
      \end{itemize}
  \end{itemize}
\end{frame}

\section{更多场景与应用：The ``How''}

% ===================== 第八部分：安全与原则 =====================
\section{安全问题与使用原则}

\begin{frame}{安全与风控：常见风险点}
  \begin{itemize}
    \item 代充风险：
      \begin{itemize}
        \item 别人帮你代充账号，往往需要验证码，相当于获得了你的 Google 账号“全县通行证”。
        \item 涉及银行卡、隐私信息，甚至存在海外黑钱、洗钱风险。
      \end{itemize}
    \item 使用别人的 API Key：
      \begin{itemize}
        \item 风险包括：被滥用、被风控封号、欠费难以追责。
        \item 内容是否会被第三方记录与分析？需要提前想清楚。
      \end{itemize}
    \item 使用 Router / 聚合代理：
      \begin{itemize}
        \item 等于让你的请求经过别人服务器中转。
        \item 问题是：\textbf{你的内容，你是否愿意给对方看？}
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{安全实践建议}
  \begin{itemize}
    \item 不要在聊天窗口直接粘贴私钥、API Key、数据库密码。
    \item 建议用环境变量形式管理：
      \begin{itemize}
        \item 如 \verb|export OPENAI_API_KEY=...|，在 Shell 中设置。
      \end{itemize}
    \item 账户充值与订阅：
      \begin{itemize}
        \item 尽量使用官方渠道，避免不明来源的“便宜代充”。
      \end{itemize}
    \item 数据最小暴露原则：
      \begin{itemize}
        \item 只上传为完成任务必须的最小数据集。
        \item 对敏感字段（用户信息、财务数据）尽量脱敏后再交给 AI。
      \end{itemize}
  \end{itemize}
\end{frame}

% ===================== 第九部分：心路历程与总结 =====================
\section{心路历程、选择与总结}

\begin{frame}{我的原则：把“一天的工作”迁移到 AI 场景}
  \begin{itemize}
    \item 雷军的愿景：让用户“把一天所有的事情都在小米手机上完成”。
    \item 类似地，斯坦福新课《CS146S: The Modern Software Developer》：
      \begin{itemize}
        \item 课程主旨：\textbf{教你在几乎不写代码的情况下进行软件开发}。
      \end{itemize}
    \item 我的迁移目标：
      \begin{itemize}
        \item 把尽可能多的工作，迁移到“与 AI 对话”的形式完成。
        \item 从写代码、改文档，到排版、测试、部署、汇报，都尽量交由 Agent。
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{模型选择：不迷信“西方月亮更圆”}
  \begin{itemize}
    \item 国产模型近两年进步巨大，性价比极高。
    \item 例子：
      \begin{itemize}
        \item Doubao-seed-code 在编码任务上，主观体验接近 Grok fast 1 / GPT 5 mini 这一档。
        \item 在某些中文视频生成场景，可灵 AI 生成的人物对话场景明显优于 Veo-3。
      \end{itemize}
    \item 加密货币领域：Binance（CZ 赵长鹏）、OKX（Star Xu）等世界级平台，均由华人创办。
    \item 结论：
      \begin{itemize}
        \item 不必盲目 FOMO 海外产品，\textbf{选适合场景 + 适合预算的组合}即可。
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{总结与行动建议}
  \begin{itemize}
    \item \textbf{从 Copilot 到 Autopilot：}
      \begin{itemize}
        \item 先让 AI 帮你“改写一段代码 / 一页文档”，再慢慢让它“改一个项目 / 一本书 / 一套工作流”。
      \end{itemize}
    \item \textbf{下周就可以做的三件事：}
      \begin{itemize}
        \item 选一个可脚本化的痛点任务（文档、脚本或报告），尝试交给 Agent 做闭环。
        \item 建一个小型个人知识库（白皮书、收藏文章、喜欢的博主字幕）。
        \item 为自己的安全策略和模型选型定一份“最低配置原则”。
      \end{itemize}
    \item 最后想说：
      \begin{itemize}
        \item AI 不会取代有架构思维和 Domain Knowledge 的人。
        \item 但会大幅拉开“会用 AI 的人”和“不会用 AI 的人”之间的生产力差距。
      \end{itemize}
  \end{itemize}
\end{frame}
