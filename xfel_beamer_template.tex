% Slide 9: 案例 - AI帮我写十万字长文Report
\begin{frame}{案例：AI 帮我写十万字长文 Report（多语种 + 排版）}
  \begin{columns}[T]
  \column{0.48\textwidth}
  \textbf{真实痛点}
  \vspace{0.2cm}
  {\scriptsize
  \begin{itemize}
    \item \textbf{长文迭代：}需要在已有白皮书基础上迭代\\
    AI 直接读 PDF → token 消耗大，理解受限
    \vspace{0.1cm}
    \item \textbf{多语种同步出版：}英文、简体中文、繁体中文\\
    传统翻译流程费时费力，改动难同步
    \vspace{0.1cm}
    \item \textbf{排版难题：}十万字长文排版、目录、交叉引用\\
    手动维护非常容易出错
    \vspace{0.1cm}
    \item \textbf{同步维护：}一处改动要三处同步，极度麻烦
  \end{itemize}}
  
  \column{0.48\textwidth}
  \textbf{我的解决方案路径}
  \vspace{0.2cm}
  {\scriptsize
  \begin{enumerate}
    \item \textbf{PDF 先转成文本}（如 .md）\\
    \textcolor{green!70!black}{→ token 减少到 1/2 \textasciitilde\ 1/20}
    \vspace{0.1cm}
    \item \textbf{"覆盖式修改"优于"重新生成"}\\
    保留原文备份，AI 在原文上直接润色/重写\\
    \textcolor{green!70!black}{→ 减少版本不一致}
    \vspace{0.1cm}
    \item \textbf{直接让 AI 操作 LaTeX}\\
    内容插入/修改、语法 debug、图表排版、编译 PDF
    \vspace{0.1cm}
    \item \textbf{脚本化 + 工作流自动化}\\
    使用大模型 API 完成长文翻译\\
    GitHub Actions 跑流程：\\
    \textcolor{blue}{\tiny 提交英文版改动 → 自动生成中/英/繁 → LaTeX 编译 → 部署}
  \end{enumerate}}
  \end{columns}
  
  \vspace{0.3cm}
  \begin{alertblock}{关键结论}
    \centering
    \textbf{整套过程，几乎可以"不写一行代码"}\\
    \small 如果你有基本的架构思维，剩下的命令、脚本都可由 AI Agent 生成与维护
  \end{alertblock}
\end{frame}

% Slide 10: 三点思考 - 如何为AI设计任务
\begin{frame}{我的三点思考：如何"为 AI 设计任务"}
  \begin{block}{\textbf{思考一：}AI 更擅长"程序化可操作"的任务}
    {\scriptsize
    \begin{itemize}
      \item \textbf{文本 vs PDF：}纯文本 (.md/.tex) 优于 PDF\\
      token 更少、没有 OCR 噪音、价格更低
      \item \textbf{记忆模式认知：}\\
      上下文记忆：只在当前会话有效\\
      训练记忆：模型训练时学到的知识，不会因对话而更新\\
      \textcolor{red}{→ 关键：现在的大模型没有真正的个人长期记忆}\\
      \textcolor{green!70!black}{→ 所以"知识库 + 脚本化"比指望模型自动记住要现实得多}
    \end{itemize}}
  \end{block}
  
  \vspace{0.2cm}
  \begin{block}{\textbf{思考二：}能被脚本化的任务，天然适合交给 Agent}
    {\scriptsize
    \begin{itemize}
      \item Word 排版：很难完全脚本化
      \item LaTeX 写书/论文：内容编写、排版、Compile 都能脚本化 \textcolor{green!70!black}{→ 非常适合 AI}
      \item PPT：我这次分享的 PPT 就是用 LaTeX (Beamer) 完成的\\
      内容、结构、设计规则都可以让 Agent 来执行
      \item \textbf{总结：}为了用好 AI，我们应该主动把任务"改造"为可脚本化的流程
    \end{itemize}}
  \end{block}
\end{frame}

% Slide 11: 思考三与Prompt套娃
\begin{frame}{思考三：Domain Knowledge 反而更重要 + Prompt 套娃技巧}
  \begin{columns}[T]
  \column{0.48\textwidth}
  \begin{block}{\small 架构思维 \& Domain Knowledge}
    \scriptsize
    \textbf{工程实现层面：}\\
    工作流（workflow）、CI/CD、YAML 等需要架构理解\\
    AI 能写 YAML，但你需要知道要什么流程
    
    \vspace{0.2cm}
    \textbf{外行 vs 从业者：}\\
    外行："我要一个网站"\\
    工程师：前端/后端/测试/运维/部署/监控
    
    \vspace{0.2cm}
    \textbf{类比：}\\
    设计师知道什么是"毛玻璃特效"\\
    PPT 设计师知道什么是"分镜"
    
    \vspace{0.2cm}
    \textbf{结论：}\\
    AI 不会自动赋予你领域知识\\
    \colorbf{Domain Knowledge 是你驾驭 AI 的"方向盘"}\\
    AI 更像是\textbf{放大有 domain knowledge 的人的生产力}
  \end{block}
  
  \column{0.48\textwidth}
  \begin{block}{\small "Prompt 套娃"技巧}
    \scriptsize
    \textbf{原始需求：}\\
    "请帮我写一个网页，加入工业风金属元素，设计高端风格。"
    
    \vspace{0.2cm}
    \textbf{套娃操作：}\\
    把这个原始需求再喂给 AI：\\
    让 AI 扮演"\colorbf{硅谷顶尖 UI 设计师 + 资深前端开发}"，\\
    帮你改写、强化这个 Prompt
    
    \vspace{0.2cm}
    \textbf{得到的强化 Prompt：}\\
    变成一段带有：
    {\tiny
    \begin{itemize}
      \item 视觉语言（工业风、拉丝金属、磨砂表面）
      \item 栅格系统规范（12-column grid）
      \item 动效规则（micro-animations、cubic-bezier）
      \item 信息架构与 HTML/CSS 结构建议
    \end{itemize}}
    直接达到"顶级科技公司视觉规范级别"的描述
  \end{block}
  
  \vspace{0.2cm}
  \begin{alertblock}{}
    \scriptsize\centering
    外行不用一开始就写出完美 Prompt\\
    让 AI 帮你写给"另一个 AI 的 Prompt" — 低成本、高回报的能力放大器
  \end{alertblock}
  \end{columns}
\end{frame}

  \begin{alertblock}{启示}
    \scriptsize
    摄影不再只是"按快门"的技术，而是：
    {\tiny
    \begin{itemize}
      \item 如何构思一个可以被 AI 延展的"视觉故事"
      \item 创作的重心，从"怎么拍"，转移到\\
      "怎么定义故事与风格约束"
    \end{itemize}}
  \end{alertblock}
  
  \column{0.48\textwidth}
  \begin{block}{安全与风控：不做"白老鼠"}
    \scriptsize
    \textbf{代充的风险：}
    {\tiny
    \begin{itemize}
      \item \textcolor{red}{风控封号：}第三方代充易触发风控
      \item \textcolor{red}{验证码问题：}相当于给了对方你的账号"万能钥匙"
      \item \textcolor{red}{海外黑钱风险：}部分"低价代充"本质是用黑卡/洗钱通道
    \end{itemize}}
    
    \vspace{0.2cm}
    \textbf{使用别人 API 的隐私问题：}
    {\tiny
    \begin{itemize}
      \item 请求内容可能被服务方记录/训练
      \item 一旦出现滥用，很难追责到源头
    \end{itemize}}
    
    \vspace{0.2cm}
    \textbf{安全实践小建议：}
    {\tiny
    \begin{itemize}
      \item API Key/私钥不要直接贴在聊天框里
      \item 更安全做法：使用环境变量\\
      \texttt{export API\_KEY="xxx"}
      \item 在代码或工具配置中引用变量
    \end{itemize}}
  \end{block}
  
  \vspace{0.2cm}
  \begin{alertblock}{}
    \scriptsize\centering
    在 AI 工具大爆发的阶段，\textbf{安全意识要比"薅羊毛"更重要}
  \end{alertblock}
  \end{columns}
\end{frame}
